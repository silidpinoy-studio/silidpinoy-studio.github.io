        const apiKey = "";
        const modelName = "gemini-2.5-flash-preview-09-2025";
        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${apiKey}`;

        let currentLessonLog = "";

        const generateBtn = document.getElementById('generateBtn');
        const generateTestBtn = document.getElementById('generateTestBtn');
        const downloadBtn = document.getElementById('downloadBtn');
        const downloadTestBtn = document.getElementById('downloadTestBtn');
        const testConfigSection = document.getElementById('testConfigSection');
        const lessonPlanOutput = document.getElementById('lessonPlanOutput');
        const testOutput = document.getElementById('testOutput');
        const messageBox = document.getElementById('messageBox');

        function showMessage(message, type = 'error') {
            messageBox.textContent = message;
            messageBox.className = 'mt-6 p-4 rounded-lg ' + (type === 'error' ? 'bg-red-100 text-red-700' : 'bg-green-100 text-green-700');
            messageBox.style.display = 'block';
            setTimeout(() => messageBox.style.display = 'none', 10000);
        }

        async function callGemini(payload) {
            let delay = 1000;
            for (let i = 0; i < 5; i++) {
                try {
                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (response.ok) {
                        const result = await response.json();
                        return result.candidates?.[0]?.content?.parts?.[0]?.text;
                    }

                    if (response.status === 429 || response.status >= 500) {
                        await new Promise(r => setTimeout(r, delay));
                        delay *= 2;
                        continue;
                    }

                    const errorData = await response.json();
                    throw new Error(errorData.error?.message || "API call failed");
                } catch (e) {
                    if (i === 4) throw e;
                    await new Promise(r => setTimeout(r, delay));
                    delay *= 2;
                }
            }
        }

        async function generateDLL() {
            lessonPlanOutput.classList.add('hidden');
            downloadBtn.classList.add('hidden');
            testConfigSection.classList.add('hidden');
            testOutput.classList.add('hidden');
            generateBtn.disabled = true;
            document.getElementById('loadingIndicator').classList.remove('hidden');
            document.getElementById('btnText').textContent = "Generating DLL...";
            
            const config = {
                subject: document.getElementById('subject').value,
                grade: document.getElementById('grade').value,
                duration: document.getElementById('duration').value,
                topic: document.getElementById('topic').value,
                cs: document.getElementById('contentStandard').value,
                ps: document.getElementById('performanceStandard').value,
                lc: document.getElementById('learningCompetencyCode').value
            };

            const systemPrompt = "You are a professional curriculum developer for the Philippine Department of Education. Generate structured Daily Lesson Logs in JSON format based on specific standards.";
            const userPrompt = `Generate a ${config.duration}-day DepEd DLL for ${config.subject} ${config.grade}. 
                Weekly Topic: ${config.topic || 'Suggest a comprehensive topic'}.
                Content Standard: ${config.cs}.
                Performance Standard: ${config.ps}.
                Learning Competency: ${config.lc}.
                IMPORTANT: Ensure the 'C_LearningCompetencyCode' field contains the COMPLETE text provided.`;

            const payload = {
                contents: [{ parts: [{ text: userPrompt }] }],
                systemInstruction: { parts: [{ text: systemPrompt }] },
                generationConfig: {
                    responseMimeType: "application/json",
                    responseSchema: {
                        type: "OBJECT",
                        properties: {
                            DailyLessons: {
                                type: "ARRAY",
                                items: {
                                    type: "OBJECT",
                                    properties: {
                                        Day: { type: "STRING" },
                                        A_ContentStandard: { type: "STRING" },
                                        B_PerformanceStandard: { type: "STRING" },
                                        C_LearningCompetencyCode: { type: "STRING" },
                                        D_LearningObjectives: { type: "STRING" },
                                        II_Content: { type: "STRING" },
                                        III_LearningResources: { type: "OBJECT", properties: { References: { type: "STRING" }, OtherResources: { type: "STRING" } } },
                                        III_Procedures: { type: "OBJECT", properties: { A_Review: { type: "STRING" }, B_Purpose: { type: "STRING" }, C_Examples: { type: "STRING" }, D_Concepts1: { type: "STRING" }, E_Concepts2: { type: "STRING" }, F_Mastery: { type: "STRING" }, G_Application: { type: "STRING" }, H_Generalization: { type: "STRING" }, I_Evaluation: { type: "STRING" }, J_Remediation: { type: "STRING" } } }
                                    }
                                }
                            }
                        }
                    }
                }
            };

            try {
                const text = await callGemini(payload);
                const data = JSON.parse(text);
                currentLessonLog = text;
                
                renderDLLTable(data);
                downloadBtn.classList.remove('hidden');
                testConfigSection.classList.remove('hidden');
                showMessage("Lesson Log generated! Assessment options available below.", "success");
            } catch (err) {
                showMessage(err.message);
            } finally {
                generateBtn.disabled = false;
                document.getElementById('loadingIndicator').classList.add('hidden');
                document.getElementById('btnText').textContent = "Generate Lesson Log";
            }
        }

        async function generateTest() {
            const selectedTypes = Array.from(document.querySelectorAll('.test-checkbox:checked')).map(cb => {
                const count = cb.closest('.test-type-card').querySelector('.test-count').value;
                return { type: cb.dataset.type, count: count };
            });

            if (selectedTypes.length === 0) {
                showMessage("Pumili ng kahit isang uri ng pagsusulit.");
                return;
            }

            testOutput.classList.add('hidden');
            downloadTestBtn.classList.add('hidden');
            generateTestBtn.disabled = true;
            document.getElementById('testLoadingIndicator').classList.remove('hidden');
            document.getElementById('testBtnText').textContent = "Bumubuo ng Pagsusulit...";

            const userPrompt = `Base sa sumusunod na Lesson Log content:
                "${currentLessonLog}"
                
                Gumawa ng isang pagsusulit (Test Paper) na may mga sumusunod na bahagi:
                ${selectedTypes.map(t => `- ${t.count} items para sa ${t.type}`).join('\n')}
                
                Format ng output:
                1. Pangalan ng Paaralan: ${document.getElementById('school').value}
                2. Subject: ${document.getElementById('subject').value}
                3. Grade: ${document.getElementById('grade').value}
                4. Quarter: ${document.getElementById('quarter').value}
                5. Malinaw na panuto (instructions) para sa bawat bahagi.
                6. Lagyan ng "Answer Key" sa pinakahuling bahagi ng papel.
                
                Output ONLY the plain text content of the test paper.`;

            try {
                const text = await callGemini({ contents: [{ parts: [{ text: userPrompt }] }] });
                testOutput.textContent = text;
                testOutput.classList.remove('hidden');
                downloadTestBtn.classList.remove('hidden');
                showMessage("Pagsusulit nabuo na!", "success");
            } catch (err) {
                showMessage(err.message);
            } finally {
                generateTestBtn.disabled = false;
                document.getElementById('testLoadingIndicator').classList.add('hidden');
                document.getElementById('testBtnText').textContent = "Generate Test Paper";
            }
        }

        function renderMergedRow(title, path, lessons, colSpan) {
            let parts = path.split('.');
            let val = parts.reduce((obj, key) => obj && obj[key], lessons[0]);
            return `<tr>
                <td style="border: 1pt solid #000; padding: 4px; font-weight: bold;">${title}</td>
                <td colspan="${colSpan}" style="border: 1pt solid #000; padding: 4px;">${val || ''}</td>
            </tr>`;
        }

        function renderDLLRow(title, path, lessons) {
            let cells = lessons.map(lesson => {
                let parts = path.split('.');
                let val = parts.reduce((obj, key) => obj && obj[key], lesson);
                return `<td style="border: 1pt solid #000; padding: 4px;">${val || ''}</td>`;
            }).join('');
            return `<tr><td style="border: 1pt solid #000; padding: 4px; font-weight: bold;">${title}</td>${cells}</tr>`;
        }
