import React, { useState, useEffect, useRef } from 'react';
import { 
  BookOpen, FileText, UploadCloud, FileType, CheckCircle, 
  Settings, Loader2, File, Copy, Download, Trash2, 
  AlertCircle, ChevronRight, FileDigit, Presentation,
  Sparkles, Smile, Star, Palette, Shapes, ArrowRight, Baby
} from 'lucide-react';

// API Key - Will be provided by execution environment
const apiKey = "";

// Supported Test Types for K-3
const AVAILABLE_TEST_TYPES = [
  { id: 'picture_mcq', label: 'Picture Multiple Choice (A, B, C)' },
  { id: 'happy_sad', label: 'True or False (😀 / 😢)' },
  { id: 'counting', label: 'Counting (How many 🍎?)' },
  { id: 'matching', label: 'Matching (Word to Picture)' },
  { id: 'missing_letters', label: 'Missing Letters / Spelling' },
  { id: 'drawing', label: 'Drawing Prompt (Draw a...)' },
  { id: 'reading_comp', label: 'Short Story + 3 Questions' },
  { id: 'sorting', label: 'Sorting/Grouping (e.g. Animals vs Toys)' },
  { id: 'fill_blanks', label: 'Fill in the Blanks (with word bank)' }
];

export default function App() {
  // Navigation State
  const [currentView, setCurrentView] = useState('home'); // 'home' or 'creator'

  // Application State
  const [mode, setMode] = useState('topic'); // 'topic' or 'files'
  const [topic, setTopic] = useState('');
  const [files, setFiles] = useState([]);
  
  const [config, setConfig] = useState({
    gradeLevel: '',
    subject: '',
    testName: ''
  });

  const [testTypes, setTestTypes] = useState(
    AVAILABLE_TEST_TYPES.map(type => ({ ...type, selected: false, count: 3 }))
  );

  const [generatedTest, setGeneratedTest] = useState('');
  const [isGenerating, setIsGenerating] = useState(false);
  const [errorMsg, setErrorMsg] = useState('');
  
  const fileInputRef = useRef(null);

  // Load Parsing Libraries on Mount
  useEffect(() => {
    const loadDependencies = () => {
      if (!window.pdfjsLib) {
        const script = document.createElement('script');
        script.src = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/2.16.105/pdf.min.js';
        document.head.appendChild(script);
        script.onload = async () => {
          try {
            const res = await fetch('https://cdnjs.cloudflare.com/ajax/libs/pdf.js/2.16.105/pdf.worker.min.js');
            const text = await res.text();
            const blob = new Blob([text], { type: 'text/javascript' });
            window.pdfjsLib.GlobalWorkerOptions.workerSrc = URL.createObjectURL(blob);
          } catch (e) {
            console.error('Failed to load PDF worker locally', e);
            window.pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/2.16.105/pdf.worker.min.js';
          }
        };
      }
      if (!window.mammoth) {
        const script = document.createElement('script');
        script.src = 'https://cdnjs.cloudflare.com/ajax/libs/mammoth/1.6.0/mammoth.browser.min.js';
        document.head.appendChild(script);
      }
      if (!window.PptxGenJS) {
        const script = document.createElement('script');
        script.src = 'https://cdn.jsdelivr.net/npm/pptxgenjs@3.12.0/dist/pptxgen.bundle.js';
        document.head.appendChild(script);
      }
    };
    loadDependencies();
  }, []);

  const handleConfigChange = (e) => {
    setConfig({ ...config, [e.target.name]: e.target.value });
  };

  const toggleTestType = (id) => {
    setTestTypes(prev => prev.map(t => t.id === id ? { ...t, selected: !t.selected } : t));
  };

  const updateTestTypeCount = (id, count) => {
    setTestTypes(prev => prev.map(t => t.id === id ? { ...t, count: parseInt(count) || 0 } : t));
  };

  const extractTextFromFile = async (file) => {
    return new Promise((resolve, reject) => {
      const fileType = file.name.split('.').pop().toLowerCase();

      if (['txt', 'csv', 'md'].includes(fileType)) {
        const reader = new FileReader();
        reader.onload = (e) => resolve(e.target.result);
        reader.onerror = () => resolve('Error reading text file.');
        reader.readAsText(file);
      } else if (fileType === 'pdf') {
        if (!window.pdfjsLib) return resolve("[System Warning] PDF library not fully loaded yet. Please try again.");
        const reader = new FileReader();
        reader.onload = async (e) => {
          try {
            const typedarray = new Uint8Array(e.target.result);
            const pdf = await window.pdfjsLib.getDocument(typedarray).promise;
            let text = '';
            for (let i = 1; i <= Math.min(pdf.numPages, 10); i++) {
              const page = await pdf.getPage(i);
              const content = await page.getTextContent();
              text += content.items.map(item => item.str).join(' ') + '\n';
            }
            resolve(text || "[Empty PDF or Scanned Images not supported]");
          } catch (err) {
            resolve(`[Error parsing PDF]: ${err.message}`);
          }
        };
        reader.readAsArrayBuffer(file);
      } else if (fileType === 'docx') {
        if (!window.mammoth) return resolve("[System Warning] DOCX library not fully loaded yet.");
        const reader = new FileReader();
        reader.onload = async (e) => {
          try {
            const arrayBuffer = e.target.result;
            const result = await window.mammoth.extractRawText({ arrayBuffer });
            resolve(result.value || "[Empty DOCX]");
          } catch (err) {
            resolve(`[Error parsing DOCX]: ${err.message}`);
          }
        };
        reader.readAsArrayBuffer(file);
      } else {
        resolve(`[Unsupported format: ${fileType}]. Please paste text manually or use .txt, .pdf, .docx.`);
      }
    });
  };

  const handleFileUpload = async (e) => {
    const selectedFiles = Array.from(e.target.files);
    if (!selectedFiles.length) return;

    const newFiles = selectedFiles.map(file => ({
      id: crypto.randomUUID(),
      name: file.name,
      size: file.size,
      status: 'parsing',
      text: ''
    }));

    setFiles(prev => [...prev, ...newFiles]);

    for (let i = 0; i < newFiles.length; i++) {
      const fileObj = newFiles[i];
      const actualFile = selectedFiles[i];
      
      const extractedText = await extractTextFromFile(actualFile);
      
      setFiles(prev => prev.map(f => 
        f.id === fileObj.id 
          ? { ...f, status: 'done', text: extractedText } 
          : f
      ));
    }
    
    if (fileInputRef.current) fileInputRef.current.value = '';
  };

  const removeFile = (id) => {
    setFiles(prev => prev.filter(f => f.id !== id));
  };

  const fetchGeminiCompletion = async (prompt, retries = 5) => {
    const delays = [1000, 2000, 4000, 8000, 16000];
    
    for (let attempt = 0; attempt < retries; attempt++) {
      try {
        const response = await fetch(
          `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`,
          {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              contents: [{ parts: [{ text: prompt }] }],
              systemInstruction: {
                parts: [{ text: "You are an expert Early Childhood Educator and Curriculum Developer for Kindergarten to Grade 3. Your goal is to create highly engaging, extremely simple, and visual assessments for young kids 4-9 years old." }]
              }
            })
          }
        );

        if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
        
        const data = await response.json();
        return data.candidates[0].content.parts[0].text;
      } catch (err) {
        if (attempt === retries - 1) throw err;
        await new Promise(res => setTimeout(res, delays[attempt]));
      }
    }
  };

  const handleGenerateTest = async () => {
    setErrorMsg('');
    
    if (!config.testName || !config.subject || !config.gradeLevel) {
      setErrorMsg("Please fill in Grade Level, Subject, and Name of Test! 🖍️");
      return;
    }

    const selectedTypes = testTypes.filter(t => t.selected && t.count > 0);
    if (selectedTypes.length === 0) {
      setErrorMsg("Please pick at least one fun activity and set the number of items! 🎈");
      return;
    }

    let sourceMaterial = "";
    if (mode === 'topic') {
      if (!topic.trim()) {
        setErrorMsg("Please tell us what the kids are learning about! 📚");
        return;
      }
      sourceMaterial = `Focus Topic: ${topic}`;
    } else {
      const validFiles = files.filter(f => f.status === 'done' && f.text.trim());
      if (validFiles.length === 0) {
        setErrorMsg("Please upload some lesson files! 📄");
        return;
      }
      sourceMaterial = validFiles.map(f => `--- Lesson: ${f.name} ---\n${f.text}\n`).join('\n\n');
    }

    const structureText = selectedTypes.map(t => `- ${t.label}: ${t.count} items`).join('\n');

    const prompt = `
Create a fun, visual, and highly engaging Summative Test tailored for early learners.

**Test Details:**
- Name of Test: ${config.testName}
- Subject: ${config.subject}
- Grade Level: ${config.gradeLevel} (Ensure vocabulary and concepts are strictly age-appropriate for K-3)

**Requested Test Structure:**
${structureText}

**Source Material / Context to base the test on:**
${sourceMaterial}

**CRITICAL INSTRUCTIONS FOR YOUNG LEARNERS (K-3):**
1. **Emojis as Visual Aids:** Kids this young rely heavily on pictures. You MUST aggressively use emojis next to key nouns, verbs, and answers. (e.g., "What color is the apple 🍎?")
2. **Real Icons (USE EMOJIS):** Where questions require actual counting or visual choices, DO NOT use text-based image placeholders. Instead, use EMOJIS directly as the visual icons! Emojis render perfectly as real graphics.
   - Example 1 (Counting): "Which picture shows only ONE ☝️ sun ☀️?"
     A) ☀️
     B) ☀️ ☀️
     C) ☀️ ☀️ ☀️
   - Example 2 (Matching): "Which one is a cat?"
     A) 🐶
     B) 🐱
     C) 🐰
3. **Simplicity:** Use extremely short sentences, sight words, and large spacing. Avoid complex phrasing.
4. **Fun Formatting:** Use encouraging headers like "Part 1: Let's Count! 🔢" instead of rigid "I. Multiple Choice".
5. **Answer Key:** Provide a clearly separated "Teacher's Answer Key 🔑" at the very end of the document.
6. Output everything in clean, readable Markdown.
`;

    setIsGenerating(true);
    try {
      const result = await fetchGeminiCompletion(prompt);
      setGeneratedTest(result);
      window.scrollTo({ top: 0, behavior: 'smooth' });
    } catch (err) {
      setErrorMsg("Oops! Something went wrong making the test. Please check your connection and try again! 🛠️");
    } finally {
      setIsGenerating(false);
    }
  };

  const handleCopy = () => {
    document.execCommand('copy');
    navigator.clipboard.writeText(generatedTest);
  };

  const handleDownloadWord = () => {
    const htmlContent = formatMarkdown(generatedTest).__html;
    
    const header = `<!DOCTYPE html>
      <html xmlns:o='urn:schemas-microsoft-com:office:office' 
            xmlns:w='urn:schemas-microsoft-com:office:word' 
            xmlns='http://www.w3.org/TR/REC-html40'>
      <head>
        <meta charset='utf-8'>
        <title>${config.testName || 'Kids Summative Test'}</title>
        <style>
          body { font-family: 'Comic Sans MS', 'Segoe UI Emoji', 'Apple Color Emoji', 'Arial', sans-serif; font-size: 14pt; color: #1e293b; }
          h1 { font-size: 22pt; text-align: center; font-weight: bold; margin-bottom: 16pt; color: #4338ca; }
          h2 { font-size: 18pt; font-weight: bold; margin-top: 24pt; margin-bottom: 12pt; color: #c026d3; }
          h3 { font-size: 16pt; font-weight: bold; margin-top: 16pt; margin-bottom: 8pt; color: #db2777; }
          p { margin-bottom: 12pt; line-height: 1.5; }
          li { margin-bottom: 8pt; line-height: 1.5; }
          strong { font-weight: bold; }
          em { font-style: italic; }
          img { max-width: 250px; height: auto; border: 2px solid #e2e8f0; border-radius: 8px; margin: 10px 0; }
        </style>
      </head>
      <body>`;
    const footer = "</body></html>";
    const fullHtml = header + htmlContent + footer;
    
    const blob = new Blob(['\ufeff', fullHtml], {
      type: 'application/msword'
    });
    
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    const fileName = config.testName ? config.testName.replace(/[^a-z0-9]/gi, '_').toLowerCase() : 'kids_test';
    a.download = `${fileName}.doc`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  };

  const handleDownloadPPT = async () => {
    if (!window.PptxGenJS) {
      console.warn("PowerPoint library is loading. Please try again in a few seconds.");
      return;
    }

    try {
      const pptx = new window.PptxGenJS();
      pptx.layout = 'LAYOUT_16x9';

      let slide = pptx.addSlide();
      slide.background = { color: "FEF2F2" }; // Very light red/pink
      slide.addText(config.testName || "Fun Summative Test!", { x: 1, y: 2, w: '80%', h: 1.5, fontSize: 54, bold: true, align: 'center', color: 'E11D48' });
      slide.addText(`${config.subject} • ${config.gradeLevel}`, { x: 1, y: 3.5, w: '80%', h: 1, fontSize: 32, align: 'center', color: '475569' });

      // Strip markdown asterisks but keep images out of the raw text
      const cleanText = generatedTest.replace(/\*\*/g, '').replace(/\*/g, '');
      const lines = cleanText.split('\n');
      
      let currentSection = '';
      let isGroupedSection = false;
      let currentGroupedText = [];
      let currentQuestionText = [];

      const commitQuestion = () => {
        if (currentQuestionText.length > 0) {
          let qSlide = pptx.addSlide();
          qSlide.background = { color: "FFFFFF" };
          
          // Add top border decorative color
          qSlide.addShape(pptx.ShapeType.rect, { x: 0, y: 0, w: "100%", h: 0.2, fill: { color: "3B82F6" } });
          
          qSlide.addText(currentSection, { x: 0.5, y: 0.4, w: '90%', h: 0.5, fontSize: 20, color: '64748B', bold: true });
          
          // Extract image URLs if any using simple regex
          let joinedText = currentQuestionText.join('\n');
          const imgRegex = /!\[([^\]]*)\]\(([^)]+)\)/;
          let match = joinedText.match(imgRegex);
          
          if (match) {
             const imgUrl = match[2];
             const textWithoutImg = joinedText.replace(imgRegex, '');
             qSlide.addText(textWithoutImg, { x: 0.5, y: 1.2, w: '90%', h: '30%', fontSize: 32, color: '1E293B', valign: 'top' });
             qSlide.addImage({ path: imgUrl, x: 2, y: 3, w: 4, h: 2, sizing: { type: 'contain' } });
          } else {
             qSlide.addText(joinedText, { x: 0.5, y: 1.2, w: '90%', h: '80%', fontSize: 36, color: '1E293B', valign: 'top' });
          }
          currentQuestionText = [];
        }
      };

      const commitGroup = () => {
        if (currentGroupedText.length > 0) {
          let gSlide = pptx.addSlide();
          gSlide.background = { color: "FFFFFF" };
          gSlide.addShape(pptx.ShapeType.rect, { x: 0, y: 0, w: "100%", h: 0.2, fill: { color: "8B5CF6" } });
          
          gSlide.addText(currentSection, { x: 0.5, y: 0.4, w: '90%', h: 0.5, fontSize: 28, color: '8B5CF6', bold: true });
          
          // Clean out markdown images for grouped slides to avoid clutter
          let cleanedText = currentGroupedText.join('\n').replace(/!\[([^\]]*)\]\(([^)]+)\)/g, '🖼️ [$1]');
          gSlide.addText(cleanedText, { x: 0.5, y: 1.2, w: '90%', h: '80%', fontSize: 24, color: '1E293B', valign: 'top' });
          currentGroupedText = [];
        }
      };

      for (let i = 0; i < lines.length; i++) {
        const line = lines[i];
        
        if (line.trim().startsWith('#')) {
          if (isGroupedSection) commitGroup();
          else commitQuestion();

          currentSection = line.replace(/#/g, '').trim();
          const lowerSec = currentSection.toLowerCase();
          
          isGroupedSection = lowerSec.includes('match') || lowerSec.includes('key') || lowerSec.includes('story') || lowerSec.includes('sort') || lowerSec.includes('blank');
          
          let secSlide = pptx.addSlide();
          secSlide.background = { color: "FDF4FF" }; // Very light purple
          secSlide.addText(currentSection, { x: 1, y: 2.5, w: '80%', h: 1.5, fontSize: 50, bold: true, align: 'center', color: 'C026D3' });
          continue;
        }

        if (line.trim() === '') {
           if (isGroupedSection && currentGroupedText.length > 0 && currentGroupedText[currentGroupedText.length-1] !== '') {
               currentGroupedText.push('');
           } else if (!isGroupedSection && currentQuestionText.length > 0 && currentQuestionText[currentQuestionText.length-1] !== '') {
               currentQuestionText.push('');
           }
           continue;
        }

        if (isGroupedSection) {
          currentGroupedText.push(line);
        } else {
          if (/^\d+\./.test(line.trim())) {
            commitQuestion();
            currentQuestionText.push(line);
          } else {
            currentQuestionText.push(line);
          }
        }
      }
      
      if (isGroupedSection) commitGroup();
      else commitQuestion();

      const fileName = config.testName ? config.testName.replace(/[^a-z0-9]/gi, '_').toLowerCase() : 'kids_test';
      pptx.writeFile({ fileName: `${fileName}.pptx` });
    } catch (error) {
      console.error("PPT Generation Error:", error);
    }
  };

  // Upgraded Markdown Formatter to support images and larger kid-friendly styles
  const formatMarkdown = (text) => {
    if (!text) return { __html: '' };
    
    let html = text
      // Headers
      .replace(/^### (.*$)/gim, '<h3 class="text-2xl font-bold mt-6 mb-3 text-pink-600 border-b-2 border-pink-100 pb-1">$1</h3>')
      .replace(/^## (.*$)/gim, '<h2 class="text-3xl font-extrabold mt-8 mb-4 text-purple-600 border-b-4 border-purple-200 pb-2 rounded-sm">$1</h2>')
      .replace(/^# (.*$)/gim, '<h1 class="text-4xl font-black mt-8 mb-6 text-center text-cyan-600 drop-shadow-sm">$1</h1>')
      // Images
      .replace(/!\[([^\]]+)\]\(([^)]+)\)/g, '<img src="$2" alt="$1" class="my-4 rounded-xl shadow-sm inline-block max-w-[200px] border-4 border-slate-100" />')
      // Multiple Choice Options (A, B, C) - Styled as large, kid-friendly yellow buttons!
      .replace(/^[\s*\-]*([A-E][\.\)])\s+(.*)/gim, '<div class="my-3 p-4 bg-yellow-50 border-4 border-yellow-200 rounded-2xl shadow-sm flex items-center gap-4 break-words"><span class="font-black text-amber-500 text-2xl w-10 flex-shrink-0">$1</span> <span class="text-4xl tracking-widest text-slate-800 leading-tight">$2</span></div>')
      // Bold & Italics
      .replace(/\*\*(.*?)\*\*/gim, '<strong class="text-slate-900 font-extrabold">$1</strong>')
      .replace(/\*(.*?)\*/gim, '<em class="text-slate-700">$1</em>')
      // Lists
      .replace(/^\s*-\s(.*)/gim, '<li class="ml-6 list-none mb-2 flex items-start gap-2 before:content-[\'⭐\'] before:mr-1">$1</li>')
      // Line breaks
      .replace(/\n/gim, '<br />');

    return { __html: html };
  };

  return (
    <div className="min-h-screen bg-amber-50 font-sans text-slate-800">
      {/* Print Styles */}
      <style dangerouslySetInnerHTML={{__html: `
        @media print {
          body { background: white; font-size: 14pt !important; }
          .no-print { display: none !important; }
          .print-full { width: 100% !important; max-width: none !important; padding: 0 !important; margin: 0 !important; border: none !important; box-shadow: none !important; }
          .page-break { page-break-inside: avoid; }
        }
      `}} />

      {/* Header */}
      <header className="bg-cyan-500 text-white shadow-lg sticky top-0 z-20 no-print border-b-4 border-cyan-600">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-4 flex items-center justify-between">
          <div 
            className="flex items-center gap-3 cursor-pointer hover:scale-105 transition-transform"
            onClick={() => setCurrentView('home')}
          >
            <div className="bg-white text-cyan-500 p-2 rounded-full shadow-inner">
              <Baby className="w-8 h-8" />
            </div>
            <div>
              <h1 className="text-2xl font-black tracking-wide leading-tight drop-shadow-md">K-3 Test Maker</h1>
              <p className="text-cyan-100 text-sm font-bold">Fun & Easy Assessments 🎈</p>
            </div>
          </div>
          <nav className="hidden sm:flex gap-3">
            <button 
              onClick={() => setCurrentView('home')} 
              className={`text-sm font-bold px-5 py-2.5 rounded-full transition-all shadow-sm ${currentView === 'home' ? 'bg-white text-cyan-600' : 'text-cyan-50 bg-cyan-600 hover:bg-cyan-400'}`}
            >
              Home
            </button>
            <button 
              onClick={() => setCurrentView('creator')} 
              className={`text-sm font-bold px-5 py-2.5 rounded-full transition-all shadow-sm ${currentView === 'creator' ? 'bg-white text-cyan-600' : 'text-cyan-50 bg-cyan-600 hover:bg-cyan-400'}`}
            >
              Create Test ✨
            </button>
          </nav>
        </div>
      </header>

      {/* Main Content Rendering based on View */}
      {currentView === 'home' ? (
        <main className="flex flex-col min-h-[calc(100vh-80px)]">
          {/* Hero Section */}
          <section className="bg-gradient-to-b from-amber-50 to-pink-50 border-b border-pink-100 pt-20 pb-24 px-4 sm:px-6 lg:px-8 text-center flex-1 flex flex-col justify-center items-center relative overflow-hidden">
            {/* Playful Background Elements */}
            <div className="absolute top-10 left-10 text-pink-300 opacity-50 rotate-12"><Star className="w-20 h-20 fill-current" /></div>
            <div className="absolute bottom-20 right-10 text-cyan-300 opacity-50 -rotate-12"><Shapes className="w-24 h-24 fill-current" /></div>
            <div className="absolute top-20 right-32 text-amber-300 opacity-50 rotate-45"><Smile className="w-16 h-16 fill-current" /></div>
            
            <div className="relative z-10 max-w-3xl mx-auto">
              <span className="inline-flex items-center gap-2 px-4 py-1.5 rounded-full bg-white text-pink-600 text-sm font-bold mb-8 shadow-md border-2 border-pink-100">
                <Palette className="w-5 h-5" /> Crafted for Early Learners
              </span>
              <h1 className="text-5xl md:text-6xl lg:text-7xl font-black text-slate-800 tracking-tight mb-8 leading-tight drop-shadow-sm">
                Make learning <span className="text-transparent bg-clip-text bg-gradient-to-r from-pink-500 to-purple-500">Fun & Visual!</span> 🎨
              </h1>
              <p className="text-xl md:text-2xl text-slate-600 mb-12 max-w-2xl mx-auto leading-relaxed font-medium">
                Generate highly visual, emoji-filled, and super engaging quizzes for Kindergarten to Grade 3 in seconds.
              </p>
              
              <div className="flex flex-col sm:flex-row items-center justify-center gap-5">
                <button 
                  onClick={() => setCurrentView('creator')}
                  className="w-full sm:w-auto px-10 py-5 bg-pink-500 text-white rounded-full font-black text-xl shadow-lg shadow-pink-200 hover:bg-pink-600 hover:scale-105 transition-all flex items-center justify-center gap-3 border-b-4 border-pink-700 active:border-b-0 active:translate-y-1"
                >
                  Start Creating <ArrowRight className="w-6 h-6" />
                </button>
                <button 
                  onClick={() => {
                    setMode('files');
                    setCurrentView('creator');
                  }}
                  className="w-full sm:w-auto px-10 py-5 bg-white text-purple-600 rounded-full font-black text-xl shadow-lg hover:bg-purple-50 hover:scale-105 transition-all flex items-center justify-center gap-3 border-2 border-purple-100 border-b-4 border-purple-200 active:border-b-2 active:translate-y-1"
                >
                  <UploadCloud className="w-6 h-6" /> Upload Modules
                </button>
              </div>
            </div>
          </section>

          {/* Features Section */}
          <section className="bg-white py-24 px-4 sm:px-6 lg:px-8 border-t-4 border-dashed border-cyan-200">
            <div className="max-w-7xl mx-auto">
              <div className="text-center mb-16">
                <h2 className="text-4xl font-black text-slate-800 mb-4 drop-shadow-sm">Everything a teacher needs 🎒</h2>
                <p className="text-lg font-medium text-slate-500 max-w-2xl mx-auto">Designed to save you time while creating perfectly tailored, developmentally appropriate activities.</p>
              </div>
              
              <div className="grid md:grid-cols-3 gap-8">
                {/* Feature 1 */}
                <div className="bg-amber-50 p-8 rounded-[2rem] shadow-sm border-2 border-amber-200 hover:shadow-xl hover:-translate-y-1 transition-all">
                  <div className="w-16 h-16 bg-white rounded-full flex items-center justify-center mb-6 text-amber-500 shadow-sm border-2 border-amber-100">
                    <Smile className="w-8 h-8" />
                  </div>
                  <h3 className="text-2xl font-bold text-slate-800 mb-3">Emoji & Visual Rich</h3>
                  <p className="text-slate-600 leading-relaxed font-medium">
                    The AI automatically inserts emojis and image placeholders to make reading easier for early learners.
                  </p>
                </div>

                {/* Feature 2 */}
                <div className="bg-pink-50 p-8 rounded-[2rem] shadow-sm border-2 border-pink-200 hover:shadow-xl hover:-translate-y-1 transition-all">
                  <div className="w-16 h-16 bg-white rounded-full flex items-center justify-center mb-6 text-pink-500 shadow-sm border-2 border-pink-100">
                    <Shapes className="w-8 h-8" />
                  </div>
                  <h3 className="text-2xl font-bold text-slate-800 mb-3">Age-Appropriate</h3>
                  <p className="text-slate-600 leading-relaxed font-medium">
                    Test types are tailored for kids—counting, drawing prompts, missing letters, and simple picture matching.
                  </p>
                </div>

                {/* Feature 3 */}
                <div className="bg-purple-50 p-8 rounded-[2rem] shadow-sm border-2 border-purple-200 hover:shadow-xl hover:-translate-y-1 transition-all">
                  <div className="w-16 h-16 bg-white rounded-full flex items-center justify-center mb-6 text-purple-500 shadow-sm border-2 border-purple-100">
                    <Presentation className="w-8 h-8" />
                  </div>
                  <h3 className="text-2xl font-bold text-slate-800 mb-3">Big Print Export</h3>
                  <p className="text-slate-600 leading-relaxed font-medium">
                    One-click export to Word with large 14pt fonts ready for printing, or bright colorful PowerPoint slides.
                  </p>
                </div>
              </div>
            </div>
          </section>

          {/* Footer */}
          <footer className="bg-slate-800 py-10 text-center text-slate-300 text-sm font-medium">
            <p>© {new Date().getFullYear()} K-3 Test Maker. Made with ❤️ for Teachers.</p>
          </footer>
        </main>
      ) : (
        <main className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8 flex flex-col lg:flex-row gap-8">
          
          {/* Left Column: Configuration Form */}
          <section className="w-full lg:w-[450px] flex-shrink-0 space-y-6 no-print">
            
            {/* Error Banner */}
            {errorMsg && (
              <div className="bg-red-100 border-2 border-red-300 text-red-800 px-5 py-4 rounded-2xl flex items-start gap-3 shadow-sm">
                <AlertCircle className="w-6 h-6 flex-shrink-0 mt-0.5" />
                <span className="font-bold text-sm">{errorMsg}</span>
              </div>
            )}

            {/* Card 1: Basic Details */}
            <div className="bg-white rounded-3xl shadow-sm border-2 border-purple-100 p-6">
              <h2 className="text-xl font-black text-purple-600 mb-5 flex items-center gap-3">
                <Settings className="w-6 h-6" /> Test Details 📝
              </h2>
              <div className="space-y-4">
                <div>
                  <label className="block text-sm font-extrabold text-slate-600 mb-1.5">Grade Level</label>
                  <input 
                    type="text" name="gradeLevel" value={config.gradeLevel} onChange={handleConfigChange}
                    placeholder="e.g. Kindergarten, Grade 1" 
                    className="w-full px-5 py-3 bg-purple-50 border-2 border-purple-200 rounded-xl font-medium focus:ring-4 focus:ring-purple-200 focus:border-purple-400 focus:outline-none transition-all placeholder:text-purple-300"
                  />
                </div>
                <div>
                  <label className="block text-sm font-extrabold text-slate-600 mb-1.5">Subject</label>
                  <input 
                    type="text" name="subject" value={config.subject} onChange={handleConfigChange}
                    placeholder="e.g. Math, Reading, Science" 
                    className="w-full px-5 py-3 bg-purple-50 border-2 border-purple-200 rounded-xl font-medium focus:ring-4 focus:ring-purple-200 focus:border-purple-400 focus:outline-none transition-all placeholder:text-purple-300"
                  />
                </div>
                <div>
                  <label className="block text-sm font-extrabold text-slate-600 mb-1.5">Name of Test</label>
                  <input 
                    type="text" name="testName" value={config.testName} onChange={handleConfigChange}
                    placeholder="e.g. Animals Quiz!" 
                    className="w-full px-5 py-3 bg-purple-50 border-2 border-purple-200 rounded-xl font-medium focus:ring-4 focus:ring-purple-200 focus:border-purple-400 focus:outline-none transition-all placeholder:text-purple-300"
                  />
                </div>
              </div>
            </div>

            {/* Card 2: Source Material */}
            <div className="bg-white rounded-3xl shadow-sm border-2 border-amber-100 p-6">
              <h2 className="text-xl font-black text-amber-600 mb-5 flex items-center gap-3">
                <FileText className="w-6 h-6" /> What are we testing? 🧠
              </h2>
              
              {/* Mode Toggle */}
              <div className="flex bg-amber-50 p-1.5 rounded-2xl mb-5 border border-amber-200">
                <button 
                  onClick={() => setMode('topic')}
                  className={`flex-1 py-2.5 text-sm font-extrabold rounded-xl transition-all ${mode === 'topic' ? 'bg-white text-amber-600 shadow-sm border border-amber-100' : 'text-amber-700/60 hover:text-amber-700'}`}
                >
                  Type a Topic
                </button>
                <button 
                  onClick={() => setMode('files')}
                  className={`flex-1 py-2.5 text-sm font-extrabold rounded-xl transition-all ${mode === 'files' ? 'bg-white text-amber-600 shadow-sm border border-amber-100' : 'text-amber-700/60 hover:text-amber-700'}`}
                >
                  Upload Files
                </button>
              </div>

              {/* Topic Input */}
              {mode === 'topic' && (
                <div>
                  <textarea 
                    value={topic}
                    onChange={(e) => setTopic(e.target.value)}
                    placeholder="e.g. Recognizing farm animals vs wild animals. Counting to 10..."
                    className="w-full h-40 px-5 py-4 bg-amber-50 border-2 border-amber-200 rounded-xl font-medium focus:ring-4 focus:ring-amber-200 focus:border-amber-400 focus:outline-none resize-none transition-all placeholder:text-amber-300"
                  ></textarea>
                </div>
              )}

              {/* File Upload */}
              {mode === 'files' && (
                <div className="space-y-4">
                  <div 
                    className="border-4 border-dashed border-amber-200 bg-amber-50/50 rounded-2xl p-8 text-center hover:bg-amber-100 transition-colors cursor-pointer relative"
                    onClick={() => fileInputRef.current?.click()}
                  >
                    <div className="bg-white w-14 h-14 rounded-full flex items-center justify-center mx-auto mb-3 shadow-sm">
                      <UploadCloud className="w-7 h-7 text-amber-500" />
                    </div>
                    <p className="text-base font-bold text-amber-800">Tap to upload lessons</p>
                    <p className="text-xs font-medium text-amber-600/70 mt-1">.pdf, .docx, .txt</p>
                    <input 
                      type="file" 
                      multiple 
                      ref={fileInputRef}
                      onChange={handleFileUpload} 
                      className="hidden"
                      accept=".pdf,.docx,.txt,.csv,.md"
                    />
                  </div>

                  {files.length > 0 && (
                    <div className="space-y-3 mt-4">
                      <p className="text-xs font-black text-amber-600/80 uppercase tracking-wide">Attached Materials</p>
                      {files.map(file => (
                        <div key={file.id} className="bg-white rounded-2xl border-2 border-amber-100 p-3 flex items-center justify-between">
                          <div className="flex items-center gap-3 overflow-hidden">
                            <div className="bg-amber-100 p-2 rounded-xl">
                              <FileDigit className="w-5 h-5 text-amber-600" />
                            </div>
                            <div className="overflow-hidden">
                              <p className="text-sm font-bold text-slate-700 truncate" title={file.name}>{file.name}</p>
                              {file.status === 'parsing' ? (
                                <span className="flex items-center gap-1 text-[10px] text-amber-600 font-black uppercase mt-0.5">
                                  <Loader2 className="w-3 h-3 animate-spin" /> Reading...
                                </span>
                              ) : (
                                <span className="flex items-center gap-1 text-[10px] text-emerald-500 font-black uppercase mt-0.5">
                                  <CheckCircle className="w-3 h-3" /> Ready
                                </span>
                              )}
                            </div>
                          </div>
                          <button onClick={() => removeFile(file.id)} className="text-slate-300 hover:text-red-500 p-2 rounded-xl hover:bg-red-50 transition-colors">
                            <Trash2 className="w-5 h-5" />
                          </button>
                        </div>
                      ))}
                    </div>
                  )}
                </div>
              )}
            </div>

            {/* Card 3: Test Types Selection */}
            <div className="bg-white rounded-3xl shadow-sm border-2 border-pink-100 p-6">
              <h2 className="text-xl font-black text-pink-500 mb-3 flex items-center gap-3">
                <FileType className="w-6 h-6" /> Fun Activities 🎨
              </h2>
              <p className="text-sm font-medium text-slate-500 mb-5">Pick the games & questions for the test!</p>
              
              <div className="space-y-3 max-h-80 overflow-y-auto pr-2 custom-scrollbar">
                {testTypes.map((type) => (
                  <div key={type.id} className={`flex items-center justify-between p-4 rounded-2xl border-2 transition-all ${type.selected ? 'bg-pink-50 border-pink-300 shadow-sm' : 'bg-white border-slate-100 hover:border-pink-200'}`}>
                    <label className="flex items-center gap-3 cursor-pointer flex-1">
                      <div className="relative flex items-center">
                        <input 
                          type="checkbox" 
                          checked={type.selected} 
                          onChange={() => toggleTestType(type.id)}
                          className="peer w-6 h-6 text-pink-500 bg-slate-100 border-2 border-slate-300 rounded-lg focus:ring-pink-500 focus:ring-2 cursor-pointer appearance-none checked:bg-pink-500 checked:border-pink-500 transition-all"
                        />
                        <CheckCircle className="w-4 h-4 text-white absolute left-1 pointer-events-none opacity-0 peer-checked:opacity-100 transition-opacity" />
                      </div>
                      <span className={`text-sm font-bold ${type.selected ? 'text-pink-700' : 'text-slate-600'}`}>
                        {type.label}
                      </span>
                    </label>
                    
                    {type.selected && (
                      <div className="flex items-center gap-2 flex-shrink-0 animate-fade-in bg-white px-2 py-1.5 rounded-xl border-2 border-pink-100">
                        <span className="text-xs font-black text-pink-400">Qty:</span>
                        <input 
                          type="number" 
                          min="1" max="50"
                          value={type.count}
                          onChange={(e) => updateTestTypeCount(type.id, e.target.value)}
                          className="w-12 px-1 py-0.5 text-sm font-bold text-center text-pink-700 bg-transparent border-none focus:ring-0 outline-none"
                        />
                      </div>
                    )}
                  </div>
                ))}
              </div>
            </div>

            {/* Generate Action */}
            <button 
              onClick={handleGenerateTest}
              disabled={isGenerating}
              className={`w-full py-5 rounded-3xl font-black text-xl shadow-lg transition-all flex items-center justify-center gap-3 border-b-4 ${isGenerating ? 'bg-slate-300 text-slate-500 border-slate-400 cursor-not-allowed' : 'bg-cyan-500 text-white border-cyan-700 hover:bg-cyan-400 hover:shadow-cyan-200 active:border-b-0 active:translate-y-1'}`}
            >
              {isGenerating ? (
                <><Loader2 className="w-7 h-7 animate-spin" /> Making Magic... ✨</>
              ) : (
                <>Generate Fun Test! <Sparkles className="w-6 h-6" /></>
              )}
            </button>

          </section>

          {/* Right Column: Output / Preview */}
          <section className="flex-1 min-w-0 bg-white rounded-[2rem] shadow-xl border-4 border-white ring-1 ring-slate-100 overflow-hidden flex flex-col print-full">
            
            {/* Output Toolbar */}
            <div className="bg-gradient-to-r from-cyan-50 to-blue-50 border-b-2 border-blue-100 px-6 py-4 flex flex-wrap items-center justify-between gap-4 no-print">
              <h3 className="font-black text-lg text-cyan-800 flex items-center gap-2">
                <Star className="w-5 h-5 text-yellow-400 fill-current" /> Test Preview
              </h3>
              
              {generatedTest && (
                <div className="flex flex-wrap items-center gap-3">
                  <button onClick={handleCopy} className="p-2.5 text-slate-500 hover:text-cyan-600 bg-white hover:bg-cyan-50 border-2 border-slate-200 hover:border-cyan-200 rounded-xl transition-all shadow-sm flex items-center gap-2 font-bold text-sm">
                    <Copy className="w-4 h-4" /> <span className="hidden sm:inline">Copy</span>
                  </button>
                  <button onClick={handleDownloadWord} className="p-2.5 text-indigo-600 bg-indigo-50 hover:bg-indigo-100 border-2 border-indigo-200 rounded-xl transition-all shadow-sm flex items-center gap-2 font-bold text-sm">
                    <Download className="w-4 h-4" /> <span className="hidden sm:inline">Word</span>
                  </button>
                  <button onClick={handleDownloadPPT} className="p-2.5 text-pink-600 bg-pink-50 hover:bg-pink-100 border-2 border-pink-200 rounded-xl transition-all shadow-sm flex items-center gap-2 font-bold text-sm">
                    <Presentation className="w-4 h-4" /> <span className="hidden sm:inline">PPT Slides</span>
                  </button>
                </div>
              )}
            </div>

            {/* Output Content Area */}
            <div className="p-8 lg:p-14 overflow-y-auto min-h-[700px] flex-1 bg-[url('https://www.transparenttextures.com/patterns/cubes.png')] bg-white">
              {isGenerating ? (
                <div className="h-full flex flex-col items-center justify-center text-cyan-500 space-y-6">
                  <div className="relative">
                    <Loader2 className="w-16 h-16 animate-spin text-cyan-400 absolute" />
                    <Star className="w-16 h-16 text-yellow-400 fill-current animate-pulse opacity-50" />
                  </div>
                  <p className="font-black text-xl animate-pulse tracking-wide text-cyan-600">Drawing pictures & writing questions...</p>
                </div>
              ) : generatedTest ? (
                <div 
                  className="prose prose-lg max-w-none break-words prose-headings:font-black prose-p:font-medium prose-li:font-medium"
                  dangerouslySetInnerHTML={formatMarkdown(generatedTest)}
                />
              ) : (
                <div className="h-full flex flex-col items-center justify-center text-slate-400 space-y-5 opacity-60">
                  <div className="bg-slate-100 p-6 rounded-full">
                    <Palette className="w-16 h-16 text-slate-300" />
                  </div>
                  <div className="text-center">
                    <p className="font-black text-2xl text-slate-500 mb-2">Blank Canvas!</p>
                    <p className="font-medium">Fill out the left panel to create a beautiful test.</p>
                  </div>
                </div>
              )}
            </div>
          </section>

        </main>
      )}

      {/* Global minimal styles for scrollbars and animations */}
      <style dangerouslySetInnerHTML={{__html: `
        .custom-scrollbar::-webkit-scrollbar {
          width: 8px;
        }
        .custom-scrollbar::-webkit-scrollbar-track {
          background: #fdf2f8; 
          border-radius: 8px;
        }
        .custom-scrollbar::-webkit-scrollbar-thumb {
          background: #fbcfe8; 
          border-radius: 8px;
          border: 2px solid #fdf2f8;
        }
        .custom-scrollbar::-webkit-scrollbar-thumb:hover {
          background: #f472b6; 
        }
        @keyframes fadeIn {
          from { opacity: 0; transform: translateY(-5px); }
          to { opacity: 1; transform: translateY(0); }
        }
        .animate-fade-in {
          animation: fadeIn 0.2s ease-out forwards;
        }
      `}} />
    </div>
  );
}
