        async function fetchWithRetry(url, options, maxRetries = 5) {
            const delays = [1000, 2000, 4000, 8000, 16000];
            for (let i = 0; i < maxRetries; i++) {
                try {
                    const response = await fetch(url, options);
                    if (!response.ok) {
                        throw new Error(`HTTP error! status: ${response.status}`);
                    }
                    return await response.json();
                } catch (error) {
                    if (i === maxRetries - 1) throw error;
                    await new Promise(resolve => setTimeout(resolve, delays[i]));
                }
            }
        }

        pdfUpload.addEventListener('change', async function(e) {
            const file = e.target.files[0];
            if (!file) return;
            if (file.type !== "application/pdf") {
                showError("Please upload a valid PDF file.");
                return;
            }

            hideError();
            pdfLoading.classList.remove('hidden');
            uploadedPdfText = ""; 

            try {
                const arrayBuffer = await file.arrayBuffer();
                const pdf = await pdfjsLib.getDocument(arrayBuffer).promise;
                let text = "";
                
                for (let i = 1; i <= pdf.numPages; i++) {
                    const page = await pdf.getPage(i);
                    const content = await page.getTextContent();
                    const strings = content.items.map(item => item.str);
                    text += strings.join(" ") + "\n";
                }
                
                uploadedPdfText = text;

                const apiKey = ""; 
                const extractUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
                
                const extractPayload = {
                    contents: [{ 
                        parts: [{ text: `Analyze the following Lesson Exemplar text and extract the Grade Level, Learning Competency, Curriculum Standards if present, Performance Standards if present, Learning Objectives, and any PPST/COI indicators if present. Return ONLY a valid JSON object. Do not include markdown formatting.\n\nText:\n${text.substring(0, 25000)}` }] 
                    }],
                    generationConfig: {
                        responseMimeType: "application/json",
                        responseSchema: {
                            type: "OBJECT",
                            properties: {
                                "gradeLevel": { type: "STRING" },
                                "competency": { type: "STRING" },
                                "curriculumStandards": { type: "STRING" },
                                "performanceStandards": { type: "STRING" },
                                "objectives": { type: "STRING" },
                                "ppstIndicators": { type: "STRING" }
                            }
                        }
                    }
                };

                const extractRes = await fetchWithRetry(extractUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(extractPayload)
                });

                const jsonText = extractRes.candidates?.[0]?.content?.parts?.[0]?.text;
                if (jsonText) {
                    const data = JSON.parse(jsonText);
                    if (data.gradeLevel) document.getElementById('gradeSection').value = data.gradeLevel;
                    if (data.competency) document.getElementById('competency').value = data.competency;
                    if (data.curriculumStandards) document.getElementById('curriculumStandards').value = data.curriculumStandards;
                    if (data.performanceStandards) document.getElementById('performanceStandards').value = data.performanceStandards;
                    if (data.objectives) document.getElementById('objectives').value = data.objectives;
                    if (data.ppstIndicators) document.getElementById('ppstIndicators').value = data.ppstIndicators;
                }

            } catch (err) {
                console.error("PDF Parsing Error:", err);
                showError("Failed to parse the PDF. You can still generate the lesson plan manually.");
            } finally {
                pdfLoading.classList.add('hidden');
            }
        });
